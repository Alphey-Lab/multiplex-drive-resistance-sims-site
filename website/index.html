Interactive simulator

WarningScientific-use caveat

This page runs a single in-silico stochastic model trajectory at a time. It is for exploratory simulation use only: it is not a field-performance predictor, experimental authorization tool, or regulatory decision tool. Results depend on the model assumptions and chosen parameters, and a single run does not quantify uncertainty.

This Phase-1 explorer runs the repository’s existing multiplex_drive_resistance.simulate(...) API entirely in the browser with Shinylive/Pyodide. Defaults match examples/run_examples.py.

    result_summary,

def empty_figure(message: str):
    fig, ax = plt.subplots(figsize=(9, 4))
    ax.axis("off")
    ax.text(0.5, 0.5, message, ha="center", va="center", wrap=True)
    return fig


def build_population_figure(result):
    """Build the population-size trajectory figure from a simulation result."""
    fig, ax = plt.subplots(figsize=(9, 4))
    ax.plot(result.t, result.N, label="Total", linewidth=2)
    if result.M is not None:
        ax.plot(result.t, result.M, label="Males", alpha=0.75)
    if result.F is not None:
        ax.plot(result.t, result.F, label="Females", alpha=0.75)
    ax.set_xlabel("Generation")
    ax.set_ylabel("Population size")
    ax.set_title("Population-size trajectory")
    ax.legend(loc="best")
    ax.grid(alpha=0.25)
    fig.tight_layout()
    return fig


def build_frequency_figure(result, m):
    """Build the combined allele/haplotype-frequency trajectory figure from a simulation result."""
    labels = haplotype_labels(m)
    combined = combined_frequency_trajectory(result)

    fig, ax = plt.subplots(figsize=(9, 5))
    for label, freq in zip(labels, combined):
        if np.max(freq) > 1e-4:
            ax.plot(result.t, freq, label=label)
    ax.set_xlabel("Generation")
    ax.set_ylabel("Combined frequency")
    ax.set_ylim(0, 1)
    ax.set_title("Combined allele / haplotype-frequency trajectory")
    ax.legend(loc="center left", bbox_to_anchor=(1.02, 0.5))
    ax.grid(alpha=0.25)
    fig.subplots_adjust(right=0.78)
    return fig


def _fig_to_png_bytes(fig) -> bytes:
    buf = io.BytesIO()
    fig.savefig(buf, format="png", dpi=150, bbox_inches="tight")
    plt.close(fig)
    buf.seek(0)
    return buf.read()

def server(input, output, session):
    simulation_state = reactive.value({"result": None, "error": None, "seed": None, "m": 1})

    @reactive.effect
    @reactive.event(input.run)
    def _run_simulation():
        try:
            seed_text = input.seed().strip()
            seed = None if seed_text == "" else int(seed_text)
            m = int(input.m())
            result = simulate(
                m=m,
                s=float(input.s()),
                h=float(input.h()),
                hN=float(input.hN()),
                sigma=float(input.sigma()),
                Rm=float(input.Rm()),
                K=float(input.K()),
                xD=float(input.xD()),
                yD=float(input.yD()),
                epsilon=float(input.epsilon()),
                nu=float(input.nu()),
                mu=float(input.mu()),
                beta=float(input.beta()),
                xi=float(input.xi()),
                preexisting=bool(input.preexisting()),
                T=int(input.T()),
                seed=seed,
            )
            simulation_state.set({"result": result, "error": None, "seed": seed, "m": m})
        except Exception as exc:
            simulation_state.set({"result": None, "error": str(exc), "seed": None, "m": int(input.m())})

    def _download_filename(plot_name: str) -> str:
        state = simulation_state.get()
        seed_part = str(state["seed"]) if state["seed"] is not None else "noseed"
        m_part = f"m{state['m']}"
        return f"{plot_name}_{m_part}_{seed_part}.png"

    @output
    @render.ui
    def summary():
        state = simulation_state.get()
        if state["error"] is not None:
            return ui.div(
                {"class": "alert alert-danger"},
                ui.strong("Simulation error: "),
                state["error"],
            )

        result = state["result"]
        if result is None:
            return ui.div(
                {"class": "alert alert-info"},
                "Adjust parameters, supply a seed if you want reproducible output, and click Run simulation.",
            )

        summary_data = result_summary(result)
        seed_label = "not fixed" if state["seed"] is None else str(state["seed"])
        return ui.TagList(
            ui.p(ui.strong("Resistance established: "), "Yes" if summary_data["resistance_established"] else "No"),
            ui.p(ui.strong("Final population size: "), f"{summary_data['final_population_size']:.3f}"),
            ui.p(ui.strong("Final generation reached: "), str(summary_data["final_generation"])),
            ui.p(ui.strong("Seed used: "), seed_label),
        )

    @output
    @render.plot
    def population_plot():
        state = simulation_state.get()
        if state["error"] is not None:
            return empty_figure("Fix the simulation error above to render the population-size plot.")

        result = state["result"]
        if result is None:
            return empty_figure("Run a simulation to view the population-size trajectory.")

        return build_population_figure(result)

    @output
    @render.plot
    def frequency_plot():
        state = simulation_state.get()
        if state["error"] is not None:
            return empty_figure("Fix the simulation error above to render the allele / haplotype-frequency plot.")

        result = state["result"]
        if result is None:
            return empty_figure("Run a simulation to view the combined allele / haplotype-frequency trajectory.")

        return build_frequency_figure(result, state["m"])

    @render.download(filename=lambda: _download_filename("population_trajectory"))
    def download_population_plot():
        state = simulation_state.get()
        if state["result"] is None:
            return
        fig = build_population_figure(state["result"])
        yield _fig_to_png_bytes(fig)

    @render.download(filename=lambda: _download_filename("frequency_trajectory"))
    def download_frequency_plot():
        state = simulation_state.get()
        if state["result"] is None:
            return
        fig = build_frequency_figure(state["result"], state["m"])
        yield _fig_to_png_bytes(fig)


app = App(app_ui, server)

Parameter guide

The parameters below follow the model described in Khatri & Burt (2022), PNAS 119(33):e2200567119, which analyses resistance evolution to multiplex CRISPR gene drives. The current website defaults are intended for exploration and may differ from the exact parameter values used in specific figures in that study.

Multiplex level

Input Description
m Number of guide RNAs / targeted sites in the drive construct. Singleplex (m=1) targets one site; duplex (m=2) and triplex (m=3) target two or three sites simultaneously. In the multiplex model, drive failure requires a resistance allele at every targeted site; increasing m therefore exponentially reduces the probability that a resistance haplotype is assembled by chance.

Fitness costs

Input Description
s Fitness cost associated with being homozygous for a nonfunctional resistance allele (or drive-bearing genotypes where applicable). s = 1 means fully lethal; s = 0 means no cost.
h Dominance coefficient for the fitness cost in wild-type / drive heterozygotes. h = 0 gives fully recessive costs; h = 0.5 gives additive costs.
hN Dominance coefficient for the fitness cost in wild-type / nonfunctional-resistance heterozygotes. Analogous to h but applies to the nonfunctional resistant class.
sigma Fitness cost of functional resistance when heterozygous with wild type. Functional resistant alleles retain normal gene function, so their fitness cost is generally expected to be small (sigma near 0). The fitness penalty when homozygous functional-resistant is 2 * sigma.

Population dynamics

Input Description
Rm Intrinsic rate of population growth (maximum per-generation growth factor in an empty environment). The model uses a Beverton–Holt density-dependent regulation scheme; Rm controls how strongly the population rebounds after suppression. Values near 1 give slow growth; the default (Rm = 6) represents a moderately fast-growing organism.
K Carrying capacity: the approximate equilibrium population size in the absence of drive. Larger K supports more individuals and therefore a larger pool of de novo resistant variants per generation.

Drive release frequencies

Input Description
xD Initial drive-allele frequency in males at the time of release (generation 0). A value of 0.1 corresponds to a 10 % drive frequency in males at release.
yD Initial drive-allele frequency in females at the time of release (generation 0). Can be set independently from xD; for example, releasing only drive-bearing males gives xD > 0, yD = 0.

Molecular drive parameters (per target site)

For duplex (m=2) and triplex (m=3) models, the parameters below apply independently at each target site. A resistance haplotype is only functional if all sites carry an allele that avoids drive.

Input Description
epsilon Cleavage efficiency: the probability that the drive machinery successfully cuts a susceptible (wild-type) target site in each generation. High cleavage efficiency (near 1) is required for effective drive spread.
nu NHEJ repair probability: the probability that a cleavage event at one target site is repaired by nonhomologous end joining rather than by homology-directed repair (HDR) that copies the drive. NHEJ repair produces a resistance allele (either functional or nonfunctional).
mu De novo point-mutation rate: the per-generation, per-target-site probability that a new point mutation converts a wild-type site into a resistant variant. This is a small positive number (e.g. 1 × 10⁻⁸).

Resistance variant fractions

Input Description
beta Functional fraction of NHEJ-derived resistance variants. A fraction beta of NHEJ repair outcomes produce a functional resistance allele (retaining gene function); the remaining fraction 1 − beta produce a nonfunctional allele (disrupted gene). High beta means most resistance is functional and therefore selectively neutral or only mildly costly.
xi Functional fraction of point-mutation-derived resistance variants. Analogous to beta but for de novo point mutations: a fraction xi of point mutations at a target site produce a functional resistance allele.

Burn-in and simulation duration

Input Description
preexisting Pre-release burn-in. When enabled, the simulator runs the population for an initial period without drive so that resistance variants can arise via de novo mutation and approach mutation–selection balance before the drive is introduced. The approximate duration of this burn-in depends on the model and parameters (particularly mu and K). Disabling it starts the drive release from an all-wild-type population.
T Number of generations to simulate after the drive release. The simulation may terminate earlier if the population crashes to zero.

Reproducibility

Input Description
seed Random-number seed for the stochastic simulation. Providing the same integer seed and the same parameters always reproduces the same trajectory. Leave blank for a non-deterministic run that will differ each time.

Reference: Khatri BS, Burt A (2022). PNAS 119(33):e2200567119. https://doi.org/10.1073/pnas.2200567119
